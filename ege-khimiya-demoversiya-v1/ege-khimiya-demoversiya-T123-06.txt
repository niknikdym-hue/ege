<script>
(function(){'use strict';
const root=document.getElementById('eksamio-chem-demo');if(!root)return;
const shortA=JSON.parse(document.getElementById('chem-data-short-a').textContent),shortB=JSON.parse(document.getElementById('chem-data-short-b').textContent),extended=JSON.parse(document.getElementById('chem-data-extended').textContent),all=[...shortA,...shortB,...extended];
const acceptance=JSON.parse(document.getElementById('chem-acceptance-data').textContent);
const KEY='eksamio_ege_khimiya_demo_2026_v1';const duration=210*60*1000;const variantCounts={19:2,20:2,24:2,25:3,28:2};
const $=s=>root.querySelector(s), $$=s=>[...root.querySelectorAll(s)];
const safe={get(){try{return localStorage.getItem(KEY)}catch(e){return null}},set(v){try{localStorage.setItem(KEY,v);return true}catch(e){return false}},remove(){try{localStorage.removeItem(KEY)}catch(e){}}};
const blank=()=>({started:false,finished:false,current:1,startedAt:null,endsAt:null,answers:{},flags:[],variants:{},extendedText:{},criteria:{}});
let state=blank(),timerId=null;
function load(){const raw=safe.get();if(!raw)return;try{const x=JSON.parse(raw);if(x&&typeof x==='object')state=Object.assign(blank(),x)}catch(e){state=blank()}}
function save(){safe.set(JSON.stringify(state))}
function chooseVariants(forced){Object.keys(variantCounts).forEach(n=>{const k=String(n);if(forced&&forced[k])state.variants[k]=Number(forced[k]);else if(!state.variants[k])state.variants[k]=1+Math.floor(Math.random()*variantCounts[n])})}
function taskFor(number,variant){return all.find(t=>t.number===Number(number)&&t.variant===Number(variant||1))}
function selected(number){return taskFor(number,state.variants[String(number)]||1)}
function selectedTasks(){return Array.from({length:34},(_,i)=>selected(i+1))}
function strictDigits(v){return /^\d+$/.test(v)}
function scoreShort(task,value){const v=String(value??'');const a=task.answer.canonical;const typ=task.answer.type;if(typ==='numeric_exact'){if(!/^\d+(?:[.,]\d+)?$/.test(v))return 0;const norm=x=>x.replace('.',',');return norm(v)===norm(a)?task.max_score:0}if(!strictDigits(v))return 0;if(typ==='exact_unordered_digits')return v.length===a.length&&[...v].sort().join('')===[...a].sort().join('')?task.max_score:0;if(typ==='position_partial'){if(v.length!==a.length)return 0;let wrong=0;for(let i=0;i<a.length;i++)if(v[i]!==a[i])wrong++;return wrong===0?2:wrong===1?1:0}return v===a?task.max_score:0}
function normalizeCriteria(task,values){const out=Object.assign({},values||{});if(task.number===29&&!out.equation)out.balance=false;if(task.number===34&&out._special==='single_calc_error'){(task.criterion_elements||[]).forEach(e=>out[e.id]=false)}return out}
function extScore(task,values){const clean=normalizeCriteria(task,values);if(task.number===34&&clean._special==='single_calc_error')return 3;return (task.criterion_elements||[]).reduce((s,e)=>s+(clean[e.id]?1:0),0)}
function isAnswered(number){const t=selected(number);if(t.kind==='short')return String(state.answers[number]??'')!=='';return String(state.extendedText[number]??'').trim()!==''}
function start(forced){state=blank();state.started=true;state.startedAt=Date.now();state.endsAt=Date.now()+duration;chooseVariants(forced);save();showApp();render();startTimer()}
function showApp(){$('#chem-start').style.display='none';$('#chem-results').classList.remove('is-active');$('#chem-app').classList.add('is-active')}
function showStart(){$('#chem-start').style.display='block';$('#chem-app').classList.remove('is-active');$('#chem-results').classList.remove('is-active')}
function startTimer(){clearInterval(timerId);tick();timerId=setInterval(tick,1000)}
function tick(){if(!state.started||state.finished)return;let left=Math.max(0,state.endsAt-Date.now());const h=Math.floor(left/3600000),m=Math.floor(left%3600000/60000),s=Math.floor(left%60000/1000);$('#chem-timer').textContent=[h,m,s].map(x=>String(x).padStart(2,'0')).join(':');if(left<=0)finish(true)}
function renderNav(){const nav=$('#chem-nav');nav.innerHTML='';for(let n=1;n<=34;n++){const b=document.createElement('button');b.type='button';b.className='chem-nav-btn';b.textContent=n;b.dataset.number=n;if(isAnswered(n))b.classList.add('is-answered');if(state.flags.includes(n))b.classList.add('is-flagged');if(n===state.current)b.classList.add('is-current');b.addEventListener('click',()=>{state.current=n;save();render()});nav.appendChild(b)}}
function escapeText(v){return htmlEscape(String(v??''))}function htmlEscape(v){return v.replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function render(){if(state.finished){renderResults();return}const t=selected(state.current);$('#chem-progress').textContent=`Задание ${state.current} из 34`;$('#chem-progress-sub').textContent=t.kind==='short'?`Часть 1 · максимум ${t.max_score} ${t.max_score===1?'балл':'балла'}`:`Часть 2 · максимум ${t.max_score} баллов`;renderNav();const q=$('#chem-question');const variant=t.variantLabel?`<span class="chem-variant">${t.variantLabel}</span>`:'';let answer='';if(t.kind==='short'){answer=`<div class="chem-answer-box"><label class="chem-label" for="chem-answer">Ответ</label><input class="chem-input" id="chem-answer" inputmode="${t.answer.type==='numeric_exact'?'decimal':'numeric'}" autocomplete="off" value="${escapeText(state.answers[t.number]||'')}"><div class="chem-hint">${t.input_hint}</div></div>`}else{const vals=normalizeCriteria(t,state.criteria[t.number]||{});const special=(t.special_assessment_options||[]).map(o=>`<label class="chem-check chem-special-check"><input type="checkbox" data-special-assessment="${o.id}" ${vals._special===o.id?'checked':''}><span><strong>Особая ветвь ФИПИ (${o.score} балла):</strong> ${o.text}</span></label>`).join('');const specialActive=Boolean(vals._special);const checks=t.criterion_elements.map(e=>{const disabled=specialActive||(e.depends_on&&!vals[e.depends_on]);return `<label class="chem-check ${disabled?'is-disabled':''}"><input type="checkbox" data-criterion="${e.id}" ${vals[e.id]?'checked':''} ${disabled?'disabled':''}><span>${e.text}</span></label>`}).join('');answer=`<div class="chem-answer-box"><label class="chem-label" for="chem-ext">Развёрнутый ответ</label><textarea class="chem-textarea" id="chem-ext" placeholder="Запишите решение так, как записывали бы его в бланке № 2.">${escapeText(state.extendedText[t.number]||'')}</textarea><div class="chem-self"><div class="chem-self-title">Учебная самооценка по элементам ФИПИ</div>${special}${checks}<div class="chem-self-score">Выбрано: <span id="chem-ext-score">${extScore(t,vals)}</span> из ${t.max_score}</div><div class="chem-hint">Это ориентировочная самооценка, а не официальная проверка экспертом.</div>${t.notes_html||''}</div></div>`}
q.innerHTML=`<div class="chem-question-head"><div><div class="chem-question-number">Задание ${t.number}</div>${variant}</div><div class="chem-question-meta">Источник: демоверсия ФИПИ 2026<br>печатная страница ${t.source.prompt_pdf_page}</div></div><div class="chem-prompt">${t.prompt_html}</div>${answer}<div class="chem-question-actions"><div class="chem-action-group"><button class="chem-btn chem-btn-ghost" id="chem-prev" ${t.number===1?'disabled':''}>Предыдущее</button><button class="chem-btn chem-btn-primary" id="chem-next" ${t.number===34?'disabled':''}>Следующее</button></div><div class="chem-action-group"><button class="chem-btn chem-btn-ghost" id="chem-flag">${state.flags.includes(t.number)?'Снять отметку':'Отметить'}</button><button class="chem-btn chem-btn-danger" id="chem-clear">Сбросить ответ</button></div></div>`;
$('#chem-prev').onclick=()=>go(-1);$('#chem-next').onclick=()=>go(1);$('#chem-flag').onclick=toggleFlag;$('#chem-clear').onclick=clearCurrent;
const inp=$('#chem-answer');if(inp)inp.addEventListener('input',e=>{state.answers[t.number]=e.target.value;save();renderNav()});const ta=$('#chem-ext');if(ta)ta.addEventListener('input',e=>{state.extendedText[t.number]=e.target.value;save();renderNav()});$$('[data-criterion]').forEach(c=>c.addEventListener('change',()=>{const vals=Object.assign({},state.criteria[t.number]||{});delete vals._special;vals[c.dataset.criterion]=c.checked;state.criteria[t.number]=normalizeCriteria(t,vals);save();render()}));$$('[data-special-assessment]').forEach(c=>c.addEventListener('change',()=>{const vals=Object.assign({},state.criteria[t.number]||{});vals._special=c.checked?c.dataset.specialAssessment:null;state.criteria[t.number]=normalizeCriteria(t,vals);save();render()}));}
function go(d){state.current=Math.min(34,Math.max(1,state.current+d));save();render();scrollTo({top:0,behavior:'smooth'})}
function toggleFlag(){const n=state.current;state.flags=state.flags.includes(n)?state.flags.filter(x=>x!==n):[...state.flags,n];save();render()}
function clearCurrent(){const t=selected(state.current);if(t.kind==='short')delete state.answers[t.number];else{delete state.extendedText[t.number];delete state.criteria[t.number]}save();render()}
function totals(){let sh=0,ex=0;selectedTasks().forEach(t=>{if(t.kind==='short')sh+=scoreShort(t,state.answers[t.number]||'');else ex+=extScore(t,state.criteria[t.number]||{})});return{short:sh,extended:ex,total:sh+ex}}
function finish(skipConfirm){if(!skipConfirm&&!confirm('Завершить попытку и открыть результаты?'))return;state.finished=true;save();clearInterval(timerId);renderResults()}
function renderResults(){const v=totals();$('#chem-app').classList.remove('is-active');$('#chem-start').style.display='none';const box=$('#chem-results');box.classList.add('is-active');let review='';selectedTasks().forEach(t=>{if(t.kind==='short'){const user=state.answers[t.number]||'—',score=scoreShort(t,user),ok=score===t.max_score;review+=`<article class="chem-review-card ${ok?'is-ok':'is-bad'}"><div class="chem-review-title">Задание ${t.number}${t.variantLabel?' · '+t.variantLabel:''}: ${score}/${t.max_score}</div><div class="chem-user-answer">Ваш ответ: ${escapeText(user)}</div>${t.model_answer_html}</article>`}else{const score=extScore(t,state.criteria[t.number]||{});review+=`<article class="chem-review-card"><div class="chem-review-title">Задание ${t.number}: учебная самооценка ${score}/${t.max_score}</div><div class="chem-user-answer">Ваш развёрнутый ответ сохранён в попытке.</div>${t.model_answer_html}${t.notes_html||''}</article>`}});
box.innerHTML=`<h2>Результаты попытки</h2><div class="chem-score-grid"><div class="chem-score"><b>${v.short}/36</b><span>автоматически проверенная часть</span></div><div class="chem-score"><b>${v.extended}/20</b><span>учебная самооценка развёрнутой части</span></div><div class="chem-score"><b>${v.total}/56</b><span>ориентировочная сумма</span></div></div><div class="chem-result-note"><strong>Важно:</strong> развёрнутые ответы на реальном экзамене проверяют эксперты. Показанная сумма не является официальным результатом ЕГЭ.</div><div class="chem-start-actions"><button class="chem-btn chem-btn-primary" id="chem-retry">Новая попытка</button><button class="chem-btn chem-btn-ghost" data-open-reference>Справочные материалы</button></div><div class="chem-review">${review}</div>`;$('#chem-retry').onclick=()=>{safe.remove();state=blank();showStart()};box.querySelector('[data-open-reference]').onclick=openReference;scrollTo({top:0,behavior:'smooth'})}
function openReference(){const m=$('#chem-reference');m.classList.add('is-open');document.body.style.overflow='hidden'}function closeReference(){$('#chem-reference').classList.remove('is-open');document.body.style.overflow=''}
function setRef(tab){$$('[data-ref-tab]').forEach(b=>{b.classList.toggle('chem-btn-primary',b.dataset.refTab===tab);b.classList.toggle('chem-btn-ghost',b.dataset.refTab!==tab)});$('#chem-ref-solubility').classList.toggle('is-active',tab==='solubility');$('#chem-ref-periodic').classList.toggle('is-active',tab==='periodic')}
function seedAllCorrect(){selectedTasks().forEach(t=>{if(t.kind==='short')state.answers[t.number]=t.answer.canonical;else{const v={};t.criterion_elements.forEach(e=>v[e.id]=true);state.criteria[t.number]=normalizeCriteria(t,v);state.extendedText[t.number]='Тестовый развёрнутый ответ'}});save();renderNav()}
$('#chem-start-btn').onclick=()=>start(window.__KHIMIYA_TEST_VARIANTS__||null);$('#chem-finish-top').onclick=()=>finish(false);$$('[data-open-reference]').forEach(b=>b.onclick=openReference);$$('[data-close-reference]').forEach(b=>b.onclick=closeReference);$$('[data-ref-tab]').forEach(b=>b.onclick=()=>setRef(b.dataset.refTab));$('#chem-reference').addEventListener('click',e=>{if(e.target.id==='chem-reference')closeReference()});document.addEventListener('keydown',e=>{if(e.key==='Escape')closeReference()});
window.addEventListener('DOMContentLoaded',()=>{try{const r1=JSON.parse(document.getElementById('chem-ref-solubility-data').textContent),r2=JSON.parse(document.getElementById('chem-ref-periodic-data').textContent);$('#chem-ref-solubility').src=r1.data;$('#chem-ref-periodic').src=r2.data}catch(e){console.error('Reference assets failed',e)}})
window.KhimiyaDemoTestApi={scoreShort:(n,v,input)=>scoreShort(taskFor(n,v),input),extScore:(n,values)=>extScore(taskFor(n,1),values),normalizeCriteria:(n,values)=>normalizeCriteria(taskFor(n,1),values),startWithVariants:v=>start(v),getState:()=>JSON.parse(JSON.stringify(state)),getSelectedTask:n=>selected(n),seedAllCorrect,finishNow:()=>finish(true),totals,clear:()=>{safe.remove();state=blank();showStart()},acceptance};
load();if(state.started){chooseVariants();if(state.finished)renderResults();else{showApp();render();startTimer()}}else showStart();
})();
