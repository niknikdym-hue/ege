<script>
(function () {
  var root = document.querySelector('.ep-russian-trainers');
  if (!root) return;
  var button = root.querySelector('[data-role="continue-button"]');
  if (!button) return;
  try {
    var platform = JSON.parse(localStorage.getItem('eksamio:platform-progress:v1') || 'null');
    if (!platform || !platform.trainers) return;
    var allowed = ['russian-orthoepy', 'russian-dictionary-words', 'russian-paronyms', 'russian-phraseology'];
    var recent = allowed.map(function (id) { return platform.trainers[id]; })
      .filter(function (item) { return item && item.url && item.updatedAt; })
      .sort(function (a, b) { return b.updatedAt - a.updatedAt; })[0];
    if (recent) {
      button.href = recent.url;
      button.textContent = 'Продолжить тренировку';
    }
  } catch (e) {}
})();
</script>
