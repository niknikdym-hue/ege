<script>
(function(){
'use strict';
const C=window.EKSAMIO_SOC,tasks=C.tasks.sort((a,b)=>a.n-b.n),$=s=>document.querySelector(s);
const intro=$('#soc-intro'),exam=$('#soc-exam'),finishCard=$('#soc-finish-card'),taskEl=$('#soc-task'),gridEl=$('#soc-grid');
let timerHandle=null;
function emptyState(){return {version:C.version,active:false,finished:false,current:1,startedAt:null,deadline:null,answers:{},marked:{},rubricScores:{}}}
function load(){try{const x=JSON.parse(localStorage.getItem(C.storageKey)||'null');if(x&&x.version===C.version)return Object.assign(emptyState(),x)}catch(e){}return emptyState()}
let state=load();
function setSaveWarning(show){const el=$('#soc-save-warning');if(el)el.classList.toggle('soc-hidden',!show)}
function save(){try{localStorage.setItem(C.storageKey,JSON.stringify(state));setSaveWarning(false);return true}catch(e){setSaveWarning(true);return false}}
function clearSaved(){try{localStorage.removeItem(C.storageKey);setSaveWarning(false);return true}catch(e){setSaveWarning(true);return false}}
function escapeHtml(s){return String(s==null?'':s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function rawAnswer(v){return String(v==null?'':v)}
function counts(s){const m={};for(const ch of rawAnswer(s))m[ch]=(m[ch]||0)+1;return m}
function diffCounts(user,key){const u=counts(user),k=counts(key);let extra=0,missing=0;for(const c of new Set([...Object.keys(u),...Object.keys(k)])){extra+=Math.max(0,(u[c]||0)-(k[c]||0));missing+=Math.max(0,(k[c]||0)-(u[c]||0))}return {extra,missing}}
function currentTask(){return tasks.find(t=>t.n===state.current)||tasks[0]}
function structured(t,v){if(v&&typeof v==='object')return v;if(t.interaction.type==='multiple_choice')return {type:'multiple_choice',selected:[]};return {type:'matching',values:Array(t.interaction.positions.length).fill('')}}
function codeToAnswer(t,code){const a=rawAnswer(code);if(t.interaction.type==='multiple_choice')return {type:'multiple_choice',selected:[...a].filter(x=>/^\d$/.test(x)).map(Number)};return {type:'matching',values:t.interaction.positions.map((_,i)=>/^\d$/.test(a[i]||'')?a[i]:'')}}
function answerCode(t,v){if(t.part!==1)return '';if(typeof v==='string')return v;const a=structured(t,v);if(t.interaction.type==='multiple_choice')return [...new Set((a.selected||[]).map(Number).filter(Number.isInteger))].sort((x,y)=>x-y).join('');const vals=(a.values||[]).map(x=>String(x||''));return vals.length===t.interaction.positions.length&&vals.every(Boolean)?vals.join(''):''}
function progressCode(t,v){if(t.interaction.type==='multiple_choice')return answerCode(t,v)||'—';const a=structured(t,v),vals=(a.values||[]);return vals.length===t.interaction.positions.length&&vals.every(Boolean)?vals.join(''):t.interaction.positions.map((_,i)=>vals[i]||'—').join(' ')}
function scoreShort(t,v){const a=answerCode(t,v),k=t.answer;if(!/^\d+$/.test(a))return 0;if(t.scheme==='pos1')return a===k?1:0;if(t.scheme==='set1'){const d=diffCounts(a,k);return d.extra===0&&d.missing===0?1:0}if(t.scheme==='pos2'){if(a.length!==k.length)return 0;let bad=0;for(let i=0;i<k.length;i++)if(a[i]!==k[i])bad++;return bad===0?2:bad===1?1:0}if(t.scheme==='set2'){const d=diffCounts(a,k);if(d.extra===0&&d.missing===0)return 2;if((d.extra===1&&d.missing===0)||(d.extra===0&&d.missing===1)||(d.extra===1&&d.missing===1))return 1;return 0}return 0}
function longFilled(t){const a=state.answers[t.n];return !!(a&&t.fields.some(f=>String(a[f.id]||'').trim()))}
function answerFilled(t){if(t.part===2)return longFilled(t);const a=structured(t,state.answers[t.n]);if(t.interaction.type==='matching')return (a.values||[]).length===t.interaction.positions.length&&(a.values||[]).every(Boolean);const n=(a.selected||[]).length;return t.interaction.explicitCount? n===t.interaction.explicitCount:n>0}
function startNew(){state=emptyState();state.active=true;state.startedAt=Date.now();state.deadline=Date.now()+210*60*1000;save();showExam();render()}
function resume(){state.active=true;showExam();render()}
function resetAll(){if(!confirm('Удалить сохранённую попытку и начать заново?'))return;clearSaved();state=emptyState();showIntro()}
function showIntro(){intro.classList.remove('soc-hidden');exam.classList.remove('is-active');finishCard.classList.remove('is-active');const has=state.startedAt&&!state.finished;$('#soc-continue').classList.toggle('soc-hidden',!has);$('#soc-reset-intro').classList.toggle('soc-hidden',!has)}
function showExam(){intro.classList.add('soc-hidden');finishCard.classList.remove('is-active');exam.classList.add('is-active');startTimer()}
function renderGrid(){gridEl.innerHTML=tasks.map(t=>`<button class="soc-num ${t.n===state.current?'is-current':''} ${answerFilled(t)?'is-filled':''} ${state.marked[t.n]?'is-marked':''}" data-n="${t.n}" aria-label="Задание ${t.n}">${t.n}</button>`).join('');gridEl.querySelectorAll('button').forEach(b=>b.onclick=()=>{state.current=+b.dataset.n;save();render()})}
function fieldValue(n,id){return (state.answers[n]&&state.answers[n][id])||''}
function answerPanel(t){if(t.interaction.type==='multiple_choice'){const count=t.interaction.explicitCount;return `<div class="soc-answerbox" id="soc-structured-answer"><div class="soc-answer-title">${count?`Выберите ${count} варианта`:'Отметьте все верные варианты'}</div><div class="soc-answer-hint">Цифры для бланка будут собраны автоматически${count?`; ответ считается завершённым после выбора ${count} вариантов`:''}.</div><div class="soc-code" id="soc-code">Код для бланка: ${escapeHtml(progressCode(t,state.answers[t.n]))}</div><div class="soc-answer-actions"><button class="soc-btn soc-btn--secondary" type="button" id="soc-clear-answer">Сбросить ответ</button></div></div>`}const a=structured(t,state.answers[t.n]);const rows=t.interaction.positions.map((p,i)=>{const opts=['<option value="">Не выбрано</option>'].concat(t.interaction.options.map(o=>`<option value="${o.value}" ${String((a.values||[])[i]||'')===String(o.value)?'selected':''}>${o.value}) ${escapeHtml(o.text)}</option>`)).join('');return `<div class="soc-match-row"><label class="soc-match-label" for="soc-match-${i}">${p.label})<span>${escapeHtml(p.text)}</span></label><select class="soc-match-select" id="soc-match-${i}" data-position="${i}">${opts}</select></div>`}).join('');return `<div class="soc-answerbox" id="soc-structured-answer"><div class="soc-answer-title">Установите соответствие</div><div class="soc-answer-hint">Выберите вариант для каждой буквы. Повторение цифр разрешено, если один вариант подходит нескольким позициям.</div><div class="soc-match-answer">${rows}</div><div class="soc-code" id="soc-code">Код для бланка: ${escapeHtml(progressCode(t,state.answers[t.n]))}</div><div class="soc-answer-actions"><button class="soc-btn soc-btn--secondary" type="button" id="soc-clear-answer">Сбросить ответ</button></div></div>`}
function enhanceChoiceList(t){const ol=taskEl.querySelector('ol.soc-options');if(!ol)return;const a=structured(t,state.answers[t.n]),selected=new Set((a.selected||[]).map(Number)),limit=t.interaction.explicitCount||null;const items=[...ol.children];const box=document.createElement('div');box.className='soc-choice-list';box.setAttribute('role','group');box.setAttribute('aria-label',limit?`Выберите ${limit} варианта`:'Выберите верные варианты');items.forEach((li,i)=>{const n=i+1,label=document.createElement('label');label.className='soc-choice'+(selected.has(n)?' is-selected':'');const cb=document.createElement('input');cb.type='checkbox';cb.dataset.choice=String(n);cb.checked=selected.has(n);cb.disabled=!!(limit&&selected.size>=limit&&!selected.has(n));cb.setAttribute('aria-label',`Вариант ${n}`);const num=document.createElement('span');num.className='soc-choice-number';num.textContent=String(n);const text=document.createElement('span');text.className='soc-choice-text';text.innerHTML=li.innerHTML;label.append(cb,num,text);box.append(label)});ol.replaceWith(box);box.querySelectorAll('input[type=checkbox]').forEach(cb=>cb.onchange=()=>{const cur=structured(t,state.answers[t.n]),set=new Set((cur.selected||[]).map(Number)),n=+cb.dataset.choice;if(cb.checked){if(limit&&set.size>=limit){cb.checked=false;return}set.add(n)}else set.delete(n);state.answers[t.n]={type:'multiple_choice',selected:[...set].sort((x,y)=>x-y)};save();render()})}
function bindStructured(t){if(t.interaction.type==='multiple_choice')enhanceChoiceList(t);else taskEl.querySelectorAll('.soc-match-select').forEach(sel=>sel.onchange=()=>{const a=structured(t,state.answers[t.n]),vals=[...(a.values||Array(t.interaction.positions.length).fill(''))];vals[+sel.dataset.position]=sel.value;state.answers[t.n]={type:'matching',values:vals};save();const code=$('#soc-code');if(code)code.textContent='Код для бланка: '+progressCode(t,state.answers[t.n]);renderGrid()});const clear=$('#soc-clear-answer');if(clear)clear.onclick=()=>{state.answers[t.n]=t.interaction.type==='multiple_choice'?{type:'multiple_choice',selected:[]}:{type:'matching',values:Array(t.interaction.positions.length).fill('')};save();render()}}
function render(){const t=currentTask();$('#soc-progress').style.width=((t.n/25)*100)+'%';const meta=t.part===1?`Часть 1 · до ${t.max} ${t.max===1?'балла':'баллов'}`:`Часть 2 · до ${t.max} баллов`;let answer='';if(t.part===1)answer=answerPanel(t);else answer='<div class="soc-answerbox">'+t.fields.map(f=>`<div class="soc-field"><label for="f-${t.n}-${f.id}">${f.label}</label><textarea class="soc-textarea" id="f-${t.n}-${f.id}" data-field="${f.id}">${escapeHtml(fieldValue(t.n,f.id))}</textarea><div class="soc-count" id="c-${t.n}-${f.id}"></div></div>`).join('')+'</div>';taskEl.innerHTML=`<div class="soc-task-head"><div><h2>${t.title}</h2><div class="soc-task-meta">${meta}</div></div><button class="soc-mark ${state.marked[t.n]?'is-on':''}" id="soc-mark">${state.marked[t.n]?'★ Вернуться':'☆ Вернуться позже'}</button></div><div class="soc-task-body">${t.body}</div>${answer}<div class="soc-task-nav"><button class="soc-btn soc-btn--secondary" id="soc-prev" ${t.n===1?'disabled':''}>← Назад</button><button class="soc-btn soc-btn--primary" id="soc-next">${t.n===25?'Завершить':'Далее →'}</button></div>`;$('#soc-mark').onclick=()=>{state.marked[t.n]=!state.marked[t.n];save();render()};if(t.part===1)bindStructured(t);else taskEl.querySelectorAll('textarea').forEach(ta=>{const upd=()=>{state.answers[t.n]=state.answers[t.n]||{};state.answers[t.n][ta.dataset.field]=ta.value;save();const wc=ta.value.trim()?ta.value.trim().split(/\s+/).length:0;const c=$(`#c-${t.n}-${ta.dataset.field}`);if(c)c.textContent=`Слов: ${wc}`;renderGrid()};ta.oninput=upd;upd()});$('#soc-prev').onclick=()=>{if(t.n>1){state.current=t.n-1;save();render()}};$('#soc-next').onclick=()=>{if(t.n===25)finishAttempt();else{state.current=t.n+1;save();render()}};renderGrid();window.scrollTo({top:0,behavior:'smooth'})}
function shortTotal(){return tasks.filter(t=>t.part===1).reduce((s,t)=>s+scoreShort(t,state.answers[t.n]),0)}
function allRubrics(){return tasks.filter(t=>t.part===2).flatMap(t=>t.rubrics.map(r=>({t,r})))}
function enforceRubricRules(){for(const t of tasks.filter(x=>x.part===2)){if(!longFilled(t))state.rubricScores[t.n]={}}const r=state.rubricScores[24];if(r&&r['24.1']!==undefined&&r['24.1']!==null&&r['24.1']!==''&&Number(r['24.1'])<3)r['24.2']=0}
function part2Complete(){enforceRubricRules();return tasks.filter(t=>t.part===2).every(t=>longFilled(t)&&t.rubrics.every(r=>{if(t.n===24&&r.id==='24.2'&&Number((state.rubricScores[24]||{})['24.1'])<3)return true;const v=(state.rubricScores[t.n]||{})[r.id];return v!==undefined&&v!==null&&v!==''}))}
function part2Total(){enforceRubricRules();return allRubrics().reduce((s,x)=>{if(!longFilled(x.t))return s;const value=Number((state.rubricScores[x.t.n]||{})[x.r.id])||0;return s+(x.t.n===24&&x.r.id==='24.2'&&Number((state.rubricScores[24]||{})['24.1'])<3?0:value)},0)}
function updateTotals(){const p1=shortTotal(),done=part2Complete(),p2=part2Total();$('#soc-part1-score').textContent=`${p1}/28`;$('#soc-part2-score').textContent=done?`${p2}/30`:'—/30';$('#soc-total-score').textContent='—/58'}
function showResults(){enforceRubricRules();intro.classList.add('soc-hidden');exam.classList.remove('is-active');finishCard.classList.add('is-active');const review=$('#soc-review');review.innerHTML=tasks.map(t=>{if(t.part===1){const v=answerCode(t,state.answers[t.n]),sc=scoreShort(t,state.answers[t.n]);return `<details class="soc-review-item"><summary><span>Задание ${t.n}</span><span class="soc-answer-pill ${sc===t.max?'ok':'bad'}">${sc}/${t.max}</span></summary><div class="soc-review-content"><p><strong>Ваш код:</strong> ${escapeHtml(v||'—')}</p><p><strong>Правильный ответ:</strong> ${t.answer}</p></div></details>`}const has=longFilled(t),user=t.fields.map(f=>`<p><strong>${f.label}</strong><br>${escapeHtml(fieldValue(t.n,f.id)||'—').replace(/\n/g,'<br>')}</p>`).join('');const rub=t.rubrics.map(r=>{const cur=(state.rubricScores[t.n]||{})[r.id],dependencyBlocked=t.n===24&&r.id==='24.2'&&Number((state.rubricScores[24]||{})['24.1'])!==3,disabled=!has||dependencyBlocked;const options=['<option value="">Выберите балл</option>'].concat(Object.keys(r.levels).sort((a,b)=>b-a).map(v=>`<option value="${v}" ${String(cur)===String(v)?'selected':''}>${v} — ${escapeHtml(r.levels[v])}</option>`)).join('');return `<div class="soc-rubric"><label>${r.title} · максимум ${r.max}</label><select data-task="${t.n}" data-rubric="${r.id}" ${disabled?'disabled aria-disabled="true"':''}>${options}</select>${!has?'<div class="soc-rubric-note">Самооценка недоступна: развёрнутый ответ не заполнен.</div>':''}</div>`}).join('');return `<details class="soc-review-item"><summary><span>Задание ${t.n}</span><span>${t.max} баллов</span></summary><div class="soc-review-content"><h4>Ваш ответ</h4>${user}<div class="soc-guide"><h4>Возможный ответ и требования</h4>${t.guide}</div>${rub}</div></details>`}).join('');review.querySelectorAll('select').forEach(s=>s.onchange=()=>{const n=+s.dataset.task,id=s.dataset.rubric,t=tasks.find(x=>x.n===n);if(!longFilled(t)){s.value='';return}state.rubricScores[n]=state.rubricScores[n]||{};state.rubricScores[n][id]=s.value===''?null:+s.value;if(n===24){const eligible=Number((state.rubricScores[24]||{})['24.1'])===3,sel=review.querySelector('select[data-task="24"][data-rubric="24.2"]');if(!eligible){state.rubricScores[24]['24.2']=state.rubricScores[24]['24.1']==null?null:0;if(sel)sel.value=state.rubricScores[24]['24.2']==null?'':'0'}if(id==='24.2'&&!eligible){state.rubricScores[24]['24.2']=0;s.value='0'}if(sel){sel.disabled=!eligible;sel.setAttribute('aria-disabled',String(!eligible))}}save();updateTotals()});updateTotals();window.scrollTo({top:0})}
function finishAttempt(auto=false){if(!auto&&!confirm('Завершить попытку и открыть проверку?'))return;state.active=false;state.finished=true;save();clearInterval(timerHandle);showResults()}
function startTimer(){clearInterval(timerHandle);const tick=()=>{const left=Math.max(0,(state.deadline||Date.now())-Date.now()),h=Math.floor(left/3600000),m=Math.floor((left%3600000)/60000),s=Math.floor((left%60000)/1000);$('#soc-timer').textContent=[h,m,s].map(x=>String(x).padStart(2,'0')).join(':');if(left<=0&&state.active)finishAttempt(true)};tick();timerHandle=setInterval(tick,1000)}
$('#soc-start').onclick=startNew;$('#soc-continue').onclick=resume;$('#soc-reset-intro').onclick=resetAll;$('#soc-finish').onclick=()=>finishAttempt(false);$('#soc-retry').onclick=()=>{clearSaved();state=emptyState();showIntro()};
window.EKSAMIO_SOC_TEST={tasks,state:()=>state,answerCode,scoreShort,startNew,finish:()=>finishAttempt(true),setAnswer:(n,v)=>{const t=tasks.find(x=>x.n===n);state.answers[n]=t&&t.part===1?codeToAnswer(t,v):v;save()},setLong:(n,id,v)=>{state.answers[n]=state.answers[n]||{};state.answers[n][id]=v;save()},setRubric:(n,id,v)=>{state.rubricScores[n]=state.rubricScores[n]||{};state.rubricScores[n][id]=v;save()},shortTotal,part2Total,part2Complete,render,showResults,resetStorage:()=>{clearSaved();state=emptyState()}};
if(state.finished)showResults();else if(state.active&&state.startedAt)showExam(),render();else showIntro();
})();
</script>
