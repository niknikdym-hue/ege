    tasks.sort(function(a,b){return a.number-b.number;});
    return{sources:sources,tasks:tasks};
  }
  function freshState(){return{version:4,status:"idle",current:1,answers:{},flagged:{},variantChoices:{},startedAt:null,endsAt:null,completedAt:null,shortScore:0,essayScores:{},essayZeroReasons:{},essayEligibilityConfirmed:false,essayScore:null};}
  function normalizeState(raw){
    var next=Object.assign(freshState(),raw||{});
    next.answers=next.answers&&typeof next.answers==="object"?next.answers:{};
    next.flagged=next.flagged&&typeof next.flagged==="object"?next.flagged:{};
    next.variantChoices=next.variantChoices&&typeof next.variantChoices==="object"?next.variantChoices:{};
    next.essayScores=next.essayScores&&typeof next.essayScores==="object"?next.essayScores:{};
    next.essayZeroReasons=next.essayZeroReasons&&typeof next.essayZeroReasons==="object"?next.essayZeroReasons:{};

  }
  function loadState(){try{var raw=localStorage.getItem(STORAGE_KEY);return raw?normalizeState(JSON.parse(raw)):freshState();}catch(e){return freshState();}}
  function showStorageWarning(){var x=byId("edemo-storage-warning");if(x)x.setAttribute("data-show","true");}
  function saveState(){try{localStorage.setItem(STORAGE_KEY,JSON.stringify(state));return true;}catch(e){showStorageWarning();return false;}}
  function escapeHtml(v){return String(v==null?"":v).replace(/[&<>"']/g,function(c){return{"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c];});}
  function strictDigits(v){
    if(Array.isArray(v)){if(!v.length||v.some(function(x){return !/^\d$/.test(String(x));}))return null;return v.join("");}
    var raw=String(v==null?"":v).trim();return /^\d+$/.test(raw)?raw:null;
  }
  function strictSingleWord(v){
    var raw=String(v==null?"":v).trim().toLowerCase();
    if(!/^[а-яё]+$/i.test(raw))return null;
    return raw.replace(/ё/g,"е");
  }
  function strictPhrase(v){
    var raw=String(v==null?"":v).trim().toLowerCase();
    if(!/^[а-яё]+(?: [а-яё]+)*$/i.test(raw))return null;
    return raw.replace(/ё/g,"е");
  }
  function technicalWordCount(v){
    var t=String(v||"").replace(/\u00A0/g," ");
    t=t.replace(/(?:[А-ЯЁA-Z]\.\s*){1,2}[А-ЯЁA-Z][а-яёa-z-]+/g," ИМЯ ");
    var m=t.match(/[А-ЯЁа-яёA-Za-z]+(?:-[А-ЯЁа-яёA-Za-z]+)*/g);return m?m.length:0;
  }
  function taskOptionValues(t){
    if(TASK_SCHEMAS[t.number]&&TASK_SCHEMAS[t.number].options)return TASK_SCHEMAS[t.number].options.slice();
    var html=String(t.promptHtml||""),li=html.match(/<li(?:\s[^>]*)?>/gi);
    if(li&&li.length)return Array.from({length:li.length},function(_,i){return String(i+1);});
    var markers=[],re=/\((\d+)\)/g,m;while((m=re.exec(html)))if(markers.indexOf(m[1])===-1)markers.push(m[1]);
    if(markers.length)return markers.sort(function(a,b){return Number(a)-Number(b);});
    var max=5;[t.answer].concat((t.variants||[]).map(function(v){return v.answer;})).forEach(function(a){String(a||"").split("").forEach(function(d){if(/\d/.test(d))max=Math.max(max,Number(d));});});
    return Array.from({length:max},function(_,i){return String(i+1);});
  }
  function valueDisplay(v){return Array.isArray(v)?v.filter(Boolean).join(""):String(v==null?"":v);}
  function isFilled(t,v){
    if(t.kind==="ordered_sequence")return Array.isArray(v)&&v.length===String(t.answer).length&&v.every(function(x){return /^\d$/.test(String(x));});
    if(t.kind==="unordered_digits")return Array.isArray(v)&&v.length>0;
    return String(v==null?"":v).trim().length>0;
  }
  function resolvedTask(baseTask){
    if(!baseTask.variants||!baseTask.variants.length)return baseTask;
    var idx=Number(state.variantChoices[baseTask.number]);
    if(!Number.isInteger(idx)||idx<0||idx>=baseTask.variants.length)idx=0;
    return Object.assign({},baseTask,baseTask.variants[idx],{variantIndex:idx,variantCount:baseTask.variants.length});
  }
  function ensureVariantChoices(randomizeMissing){
    tasks.forEach(function(t){
      if(!t.variants||!t.variants.length)return;
      var idx=Number(state.variantChoices[t.number]);
      if(!Number.isInteger(idx)||idx<0||idx>=t.variants.length){state.variantChoices[t.number]=randomizeMissing?Math.floor(Math.random()*t.variants.length):0;}
    });
  }
  function scoreTask(t,v){
    if(t.kind==="essay")return 0;
    if(t.kind==="word"){
      var got=strictSingleWord(v);if(got===null)return 0;
      var accepted=[t.answer].concat(t.altAnswers||[]).map(strictSingleWord);
      return accepted.indexOf(got)!==-1?t.maxScore:0;
    }
    if(t.kind==="word_compact"){
      var gotPhrase=strictPhrase(v);if(gotPhrase===null)return 0;
      var acceptedPhrases=[t.answer].concat(t.altAnswers||[]).map(strictPhrase);
      return acceptedPhrases.indexOf(gotPhrase)!==-1?t.maxScore:0;
    }
    if(t.kind==="ordered_sequence"){
      var raw=strictDigits(v),expected=String(t.answer);if(raw===null)return 0;
      if(raw===expected)return t.maxScore;if(raw.length!==expected.length)return 0;
      var diff=0;for(var i=0;i<expected.length;i++)if(raw[i]!==expected[i])diff++;
      return diff===1||diff===2?1:0;
    }
    if(t.kind==="unordered_digits"){
      var schema=TASK_SCHEMAS[t.number];
      if(schema&&schema.interaction==="sentence_numbers"){
        var gotTokens=Array.isArray(v)?v.map(String):(/^\d+$/.test(String(v==null?"":v).trim())?[String(v).trim()]:null);
        if(!gotTokens)return 0;
        var expectedTokens=schema.answerTokens.slice();
        if(gotTokens.length!==expectedTokens.length||new Set(gotTokens).size!==gotTokens.length)return 0;
        return gotTokens.slice().sort().join("|")===expectedTokens.slice().sort().join("|")?t.maxScore:0;
      }
      var rawSet=strictDigits(v),expectedSet=String(t.answer);if(rawSet===null||rawSet.length!==expectedSet.length)return 0;
      if(new Set(rawSet.split("")).size!==rawSet.length)return 0;
      return rawSet.split("").sort().join("")===expectedSet.split("").sort().join("")?t.maxScore:0;
    }
    return 0;
  }
  function correctDisplay(t){var all=[t.answer].concat(t.altAnswers||[]);return all.map(function(x){return t.number===25&&x==="всвоюочередь"?"в свою очередь":x;}).filter(function(x,i,a){return a.indexOf(x)===i;}).join("; ");}
  function formatTime(ms){var total=Math.max(0,Math.floor(ms/1000)),h=Math.floor(total/3600),m=Math.floor(total%3600/60),s=total%60;return[h,m,s].map(function(x){return String(x).padStart(2,"0");}).join(":");}

  function hasZeroReason(){return zeroReasons.some(function(r){return !!state.essayZeroReasons[r.id];});}
  function essayAssessmentComplete(){return hasZeroReason()||(state.essayEligibilityConfirmed&&criteria.every(function(c){return Object.prototype.hasOwnProperty.call(state.essayScores,c.id);}));}
  function computeEssayScore(){
    if(hasZeroReason())return 0;
    if(!state.essayEligibilityConfirmed)return null;
