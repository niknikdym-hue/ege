<script>
(function(){
'use strict';
const root=document.getElementById('eksamio-registration');
if(!root)return;

/* DEPLOYMENT CONFIGURATION: keep empty until values are explicitly approved. */
const defaults=Object.freeze({
  apiBaseUrl:'',
  personalDataConsentUrl:'',
  personalDataCheckboxText:'',
  personalDataDocumentVersion:'',
  personalDataTextVersion:'',
  marketingConsentUrl:'',
  marketingCheckboxText:'',
  marketingDocumentVersion:'',
  marketingTextVersion:'',
  authenticatedRedirectUrl:''
});
const supplied=(window.EKSAMIO_REGISTRATION_CONFIG&&typeof window.EKSAMIO_REGISTRATION_CONFIG==='object')?window.EKSAMIO_REGISTRATION_CONFIG:{};
const config=Object.freeze(Object.assign({},defaults,supplied));
const el=id=>root.querySelector('#'+id);
const email=el('er-email');
const pd=el('er-pd');
const marketing=el('er-marketing');
const beginForm=el('er-begin-form');
const verifyForm=el('er-verify-form');
const beginButton=el('er-begin');
const verifyButton=el('er-verify');
const codeStep=el('er-code-step');
const codeInput=el('er-code');
const statusBox=el('er-status');
const configWarning=el('er-config-warning');
let challengeId='';
let clientRequestId='';
let beginLocked=false;

function isVersion(value){return /^[A-Za-z0-9][A-Za-z0-9._:-]{0,127}$/.test(String(value||''));}
function isLegalUrl(value){
  const raw=String(value||'').trim();
  if(raw.startsWith('/')&&!raw.startsWith('//'))return true;
  try{return new URL(raw).protocol==='https:';}catch(_){return false;}
}
function apiBase(value){
  try{const url=new URL(String(value||'').trim());if(url.protocol!=='https:')return '';return url.href.replace(/\/$/,'');}catch(_){return '';}
}
function redirectTarget(value){
  const raw=String(value||'').trim();
  if(!raw)return '';
  if(raw.startsWith('/')&&!raw.startsWith('//'))return raw;
  try{const url=new URL(raw);return url.protocol==='https:'?url.href:'';}catch(_){return '';}
}
function validConfig(){
  return Boolean(
    apiBase(config.apiBaseUrl)&&
    isLegalUrl(config.personalDataConsentUrl)&&
    String(config.personalDataCheckboxText||'').trim()&&
    isVersion(config.personalDataDocumentVersion)&&
    isVersion(config.personalDataTextVersion)&&
    isLegalUrl(config.marketingConsentUrl)&&
    String(config.marketingCheckboxText||'').trim()&&
    isVersion(config.marketingDocumentVersion)&&
    isVersion(config.marketingTextVersion)
  );
}
function showStatus(kind,text){statusBox.dataset.kind=kind;statusBox.textContent=text;statusBox.dataset.open='true';}
function clearStatus(){statusBox.textContent='';statusBox.dataset.open='false';}
function setBusy(button,busy){button.disabled=busy||!validConfig();button.setAttribute('aria-busy',busy?'true':'false');}
function newRequestId(){
  const raw=(window.crypto&&typeof window.crypto.randomUUID==='function')?window.crypto.randomUUID():String(Date.now())+'-'+Math.random().toString(36).slice(2,12);
  return 'registration:'+raw.replace(/[^A-Za-z0-9._:-]/g,'').slice(0,145);
}
function resetIdempotency(){if(!beginLocked)clientRequestId='';}
function publicError(payload,fallback){
  if(payload&&typeof payload==='object'&&typeof payload.error==='string')return payload.error;
  return fallback;
}
async function post(path,payload){
  const base=apiBase(config.apiBaseUrl);
  if(!base)throw new Error('CONFIGURATION_REQUIRED');
  const response=await fetch(base+path,{method:'POST',credentials:'include',headers:{'Content-Type':'application/json','Accept':'application/json'},body:JSON.stringify(payload)});
  let data=null;
  try{data=await response.json();}catch(_){data=null;}
  return {response,data};
}

function applyConfig(){
  const configured=validConfig();
  if(config.personalDataCheckboxText)el('er-pd-text').textContent=String(config.personalDataCheckboxText);
  if(config.marketingCheckboxText)el('er-marketing-text').textContent=String(config.marketingCheckboxText);
  if(isLegalUrl(config.personalDataConsentUrl))el('er-pd-link').href=String(config.personalDataConsentUrl);
  if(isLegalUrl(config.marketingConsentUrl))el('er-marketing-link').href=String(config.marketingConsentUrl);
  beginButton.disabled=!configured;
  configWarning.hidden=configured;
  if(!configured){pd.disabled=true;marketing.disabled=true;email.disabled=true;}
}

[email,pd,marketing].forEach(node=>node.addEventListener('input',resetIdempotency));
[pd,marketing].forEach(node=>node.addEventListener('change',resetIdempotency));

beginForm.addEventListener('submit',async function(event){
  event.preventDefault();
  clearStatus();
  if(!validConfig()){showStatus('error','Регистрация пока не настроена.');return;}
  if(!email.checkValidity()){showStatus('error','Укажите корректный email.');email.focus();return;}
  if(!pd.checked){showStatus('error','Для регистрации требуется отдельное согласие на обработку персональных данных.');pd.focus();return;}
  if(!clientRequestId)clientRequestId=newRequestId();
  const payload={
    email:email.value.trim(),
    personal_data_consent:true,
    marketing_consent:Boolean(marketing.checked),
    personal_data_document_version:String(config.personalDataDocumentVersion),
    personal_data_text_version:String(config.personalDataTextVersion),
    marketing_document_version:String(config.marketingDocumentVersion),
    marketing_text_version:String(config.marketingTextVersion),
    client_request_id:clientRequestId
  };
  setBusy(beginButton,true);
  try{
    const result=await post('/v1/registration/begin',payload);
    if(result.response.status!==202||!result.data||result.data.status!=='CHALLENGE_SENT'||typeof result.data.challenge_id!=='string'){
      showStatus('error',publicError(result.data,'Не удалось отправить код. Повторите попытку.'));
      return;
    }
    challengeId=result.data.challenge_id;
    beginLocked=true;
    email.readOnly=true;pd.disabled=true;marketing.disabled=true;
    codeStep.dataset.open='true';
    showStatus('info','Код отправлен. Завершите подтверждение email.');
    codeInput.focus();
  }catch(error){
    showStatus('error',error&&error.message==='CONFIGURATION_REQUIRED'?'Регистрация пока не настроена.':'Сервис регистрации временно недоступен. Повторите попытку.');
  }finally{
    beginButton.disabled=beginLocked||!validConfig();
    beginButton.setAttribute('aria-busy','false');
  }
});

verifyForm.addEventListener('submit',async function(event){
  event.preventDefault();
  clearStatus();
  if(!validConfig()||!challengeId){showStatus('error','Сначала запросите код подтверждения.');return;}
  const code=codeInput.value.trim();
  if(!code){showStatus('error','Введите код подтверждения.');codeInput.focus();return;}
  setBusy(verifyButton,true);
  try{
    const result=await post('/v1/registration/verify',{challenge_id:challengeId,code:code});
    if(result.response.status!==200||!result.data||result.data.status!=='AUTHENTICATED'){
      showStatus('error',publicError(result.data,'Код не подтверждён. Проверьте его и попробуйте ещё раз.'));
      return;
    }
    verifyButton.disabled=true;codeInput.disabled=true;
    showStatus('success','Email подтверждён. Вы вошли в аккаунт ученика.');
    window.dispatchEvent(new CustomEvent('eksamio:authenticated',{detail:{status:'AUTHENTICATED'}}));
    const target=redirectTarget(config.authenticatedRedirectUrl);
    if(target)window.location.assign(target);
  }catch(_){
    showStatus('error','Сервис подтверждения временно недоступен. Повторите попытку.');
  }finally{
    if(!codeInput.disabled)setBusy(verifyButton,false);
  }
});

applyConfig();
})();
</script>
</div>
